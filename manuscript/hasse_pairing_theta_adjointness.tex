\subsection{Hasse duality and the Eisenstein bridge}

Let $p\geq5$ and put $P=p(p-1)$.  We first work on a compactified Katz
modular curve $X$ with auxiliary full level $3$.  Let $\omega$ be the Hodge
bundle, and let $C$ be the cusp divisor.  Let
$A\in H^0(X,\omega^{p-1})$ be the Hasse invariant.  Its zero divisor $S$ is
reduced and disjoint from $C$.  Put $Z=pS$.  Thus $Z$ is the Cartier divisor
cut out by $A^p$.  For every integer $r$, define
\[
 \mathcal Q_r:=\omega^r/A^p\omega^{r-P}
              \simeq \omega^r\vert_Z,
 \qquad Q_r:=H^0(Z,\mathcal Q_r).
\]
All schemes, line bundles, and sections in this subsection are over
$\mathbb F_p$ unless we state another base.  Let $\pi$ denote the tame
full-level cover of the level-one compactified moduli stack.  The following
standard descent fact records exactly how we use this cover.

\begin{lemma}[Faithful tame pullback]\label{lem:tame-pullback-faithful}
Let $\mathcal L$ be a line bundle on the level-one stack, or on a closed
subscheme obtained from it by base change.  Pullback by $\pi$ is injective on
global sections.  If the cover is Galois with group $G$, then its image is the
$G$-invariant subspace.  A $G$-invariant perfect pairing upstairs restricts
to a perfect pairing on the invariant subspaces.
\end{lemma}

\begin{proof}
Write $V$ and $W$ for the two finite-dimensional spaces in the pairing.
The cover is finite, flat, and surjective, and its degree is prime to $p$.
The trace satisfies
$\operatorname{Tr}_{\pi}\circ\pi^*=(\deg\pi)\operatorname{id}$.
Since $\deg\pi$ is a unit in $\mathbb F_p$, pullback is injective.  The same
argument applies after base change to a closed subscheme, including the
thickened Hasse divisor.  For a Galois cover, exactness of the Reynolds
operator identifies downstairs sections with invariants.  To check
perfectness, let $v\in V^G$ be nonzero and choose $w\in W$ with
$\langle v,w\rangle\ne0$.  Replace $w$ by its Reynolds average
\[
 \overline w=\frac{1}{|G|}\sum_{g\in G}gw.
\]
The pairing is $G$-invariant and $v$ is fixed, so
$\langle v,\overline w\rangle=\langle v,w\rangle\ne0$.  The same argument
with the two factors reversed proves nondegeneracy on both invariant
subspaces.  This proves the assertion about perfect pairings.
\end{proof}

All constructions below commute with $G$.  One can alternatively state them
directly on the tame compactified moduli stack.

\begin{proposition}[Natural perfect Hasse-quotient pairing]
\label{prop:hasse-perfect-pairing}
For every $r$ there is a natural perfect pairing
\[
 \langle\ ,\ \rangle_r:
 Q_r\times Q_{P+2-r}\longrightarrow \mathbb F_p.
\]
It is the product pairing followed by the trace map on the finite
Gorenstein scheme $Z$.
\end{proposition}

\begin{proof}
Because $Z$ is an effective Cartier divisor on the smooth curve $X$, its
dualizing sheaf is
\[
 \omega_Z\simeq\bigl(\omega_X\otimes\mathcal O_X(Z)\bigr)\vert_Z.
\]
Kodaira--Spencer gives $\omega_X(C)\simeq\omega^2$.  We also have
$\mathcal O_X(Z)\simeq\omega^P$.  The canonical section of
$\mathcal O_X(C)$ is nonzero on $Z$ because $C\cap Z=\varnothing$.  Hence
\[
 \omega_Z\simeq\omega^{P+2}\vert_Z.
\]
Multiplication now gives
\[
 \mathcal Q_r\otimes\mathcal Q_{P+2-r}\longrightarrow\omega_Z.
\]
Apply the trace map
$\operatorname{Tr}_Z:H^0(Z,\omega_Z)\to\mathbb F_p$ to the resulting global
section.  This gives the displayed pairing.  Duality for the finite
Gorenstein scheme $Z$ identifies $\mathcal Q_{P+2-r}$ with
$\mathcal Hom_Z(\mathcal Q_r,\omega_Z)$.  Therefore the pairing is perfect.
The standard Kodaira--Spencer isomorphism and canonical cusp section fix
the normalization used here.  Rescaling either standard choice rescales
the pairing by a nonzero scalar and does not affect perfectness or the
adjointness statement below.  The level-one statement follows from
Lemma~\ref{lem:tame-pullback-faithful}.
\end{proof}

Let $\Theta$ denote the Katz theta operator in characteristic $p$
\cite{Katz1977},
normalized by its $q$-expansion
\[
 \Theta\!\left(\sum a_nq^n\right)=\sum n a_nq^n.
\]
It raises weight by $p+1$, satisfies the product rule, and satisfies
$\Theta(A)=0$.  Hence it induces
\[
\Theta:Q_r\longrightarrow Q_{r+p+1}.
\]

\begin{lemma}[Trace and tame Igusa residues]
\label{lem:trace-igusa-residue}
Let $s\in S$, and let $t$ be a local equation for $S$ at $s$.  After a finite
extension of the residue field, the completed Igusa cover has a parameter
$u$ with $t=u^{p-1}$.  Suppose that a local section of $\omega_Z$ is
represented by
\[
 F(t)\frac{dt}{t^p}.
 \]
Its local Grothendieck trace is the coefficient of $t^{p-1}$ in $F(t)$, up to
the fixed unit from the chosen trivialization.  Its pullback has residue
$(p-1)$ times this coefficient.  Therefore an exact pullback has local trace
zero.
\end{lemma}

\begin{proof}
For the Cartier divisor $t^p=0$, the usual residue description of
duality identifies the local trace with
\[
 \operatorname{res}_{t=0}F(t)\frac{dt}{t^p}.
\]
Only the term $t^{p-1}$ contributes.  Pulling back by $t=u^{p-1}$ gives
\[
 (p-1)F(u^{p-1})u^{p-2-p(p-1)}\,du.
\]
Its coefficient of $u^{-1}du$ is $(p-1)$ times the coefficient of
$t^{p-1}$ in $F(t)$.  Since $p-1$ is nonzero in $\mathbb F_p$, a zero
residue after pullback gives a zero local trace.  The finite residue-field
extension does not change this calculation.  This follows from transitivity
of the field trace.
\end{proof}

\begin{proposition}[Theta adjointness]
\label{prop:theta-adjointness}
If $x\in Q_r$ and $y\in Q_{P+1-p-r}$, then
\[
 \langle\Theta x,y\rangle_{r+p+1}
   =-\langle x,\Theta y\rangle_r.
\]
More generally, for $m\geq0$ and
$y\in Q_{P+2-r-m(p+1)}$,
\[
 \langle\Theta^m x,y\rangle_{r+m(p+1)}
   =(-1)^m\langle x,\Theta^m y\rangle_r.
\]
\end{proposition}

\begin{proof}
The two products in the first formula have weight $P+2$.  Thus both traces
are defined.  The product rule gives
\[
 (\Theta x)y+x(\Theta y)=\Theta(xy).
\]
Notice that $h:=xy$ has weight
\[
 P+1-p=(p-1)^2.
\]
Let $a$ be the canonical section on the Igusa curve, normalized by
$a^{p-1}=A$.  After pullback, division by $a^{(p-1)^2}$ turns a local lift
of $h$ into a local function.  Under Kodaira--Spencer, the meromorphic
differential represented by $\Theta h/A^p$ is the differential of this
function.  This is the local form of the theta calculation in
\cite[Proposition~5.8]{Gross1990}; the proposition states the same identity
for graded Fourier expansions.

We apply this formula to any local lift of $h$ from $Z$ to $X$.  The answer
does not depend on the lift.  Indeed, another lift differs by $A^p g$.  Its
theta derivative differs by $A^p\Theta g$, which is zero on $Z$.

Now apply Lemma~\ref{lem:trace-igusa-residue} at each supersingular point.
Each local component of $\operatorname{Tr}_Z(\Theta h)$ is zero.  The sum of
the local traces is therefore zero.  Hence
$\operatorname{Tr}_Z(\Theta(xy))=0$.  The product rule gives the first
identity.  Induction on $m$ gives the second identity.  At each step, the
weight of $y$ decreases by $p+1$.
There is no cusp contribution: $Z$ is supported on $S$, and
$S\cap C=\varnothing$.
\end{proof}

The weight relation relevant to the Tate complement contains a further
shift of $p(p+1)$.  Let
\[
 B:=E_{p+1}\pmod p\in H^0(X,\omega^{p+1}).
\]

\begin{lemma}[The Eisenstein bridge is a unit]
\label{lem:eisenstein-unit}
The forms $A$ and $B$ have no common zero.  Hence $B\vert_Z$ is a unit and
multiplication by $B^p$ gives an isomorphism
\[
 B^p:Q_r\xrightarrow{\ \sim\ }Q_{r+p(p+1)}
\]
for every $r$.
\end{lemma}

\begin{proof}
The Hasse invariant has a simple zero at every point of $S$.  Let
$\partial_k=\Theta-kE_2/12$ be the Serre derivative in weight $k$.  The
$q$-expansion of $A$ is $1$, so $\Theta(A)=0$ in characteristic $p$.
Also $E_2\equiv E_{p+1}=B\pmod p$.  Hence the $q$-expansion principle gives
\[
 12\partial_{p-1}A=B
 \qquad\text{in characteristic }p.
\]
At a zero of $A$, the connection term is a multiple of $A$ and vanishes.
Thus $\partial_{p-1}A$ is the ordinary derivative of a local equation for
$S$, up to a nonzero change of trivialization.  It is nonzero because the
zero is simple.  Therefore $B$ does not vanish on $S$.  Its restriction to
the Artinian thickening $Z$ is consequently invertible, and so is $B^p$.
\end{proof}

\begin{theorem}[Perfect Tate-complement pairing]
\label{thm:tate-complement-pairing}
For every $r$ the formula
\[
 [x,z]_r:=\langle x,B^{-p}z\rangle_r
\]
defines a natural perfect pairing
\[
 [\ ,\ ]_r:
 Q_r\times Q_{P+2+p(p+1)-r}\longrightarrow\mathbb F_p.
\]
Under this pairing theta is skew-adjoint.  More precisely, if
\[
 z\in Q_{P+1-p+p(p+1)-r},
\]
then
\[
 [\Theta x,z]_{r+p+1}=-[x,\Theta z]_r.
\]
The same identity iterates with the sign $(-1)^m$ for $\Theta^m$.
\end{theorem}

\begin{proof}
Lemma~\ref{lem:eisenstein-unit} identifies the second factor with
$Q_{P+2-r}$.  Perfectness now follows from
Proposition~\ref{prop:hasse-perfect-pairing}.  Theta is a derivation in
characteristic $p$, so
\[
 \Theta(B^p)=pB^{p-1}\Theta(B)=0,
 \qquad
 \Theta(B^{-p})=0
\]
on $Z$.  Proposition~\ref{prop:theta-adjointness} therefore gives
\[
 \begin{aligned}
 {}[\Theta x,z]_{r+p+1}
 &=\langle\Theta x,B^{-p}z\rangle_{r+p+1}\\
 &=-\langle x,\Theta(B^{-p}z)\rangle_r\\
 &=-\langle x,B^{-p}\Theta z\rangle_r
 =-[x,\Theta z]_r.
 \end{aligned}
\]
Iteration proves the final assertion.
\end{proof}

Now let
\[
 w_-=k+2i-P,
 \qquad
 w_+=k+2i^\dagger+P,
 \qquad
 i+i^\dagger=p^2+1-k.
\]
Then
\[
 w_-+w_+=2p^2+2=P+2+p(p+1),
\]
and therefore
\[
 [\ ,\ ]_{w_-}:Q_{w_-}\times Q_{w_+}\longrightarrow\mathbb F_p
\]
is a perfect pairing under which theta is skew-adjoint.  This is the pairing
at the two candidate-weight indices.  The factor $B^p$ has weight $p(p+1)$.
It is the same factor that occurs in the leading-block formula for the $D=2$
obstruction.  Serre duality alone pairs weights whose sum is $P+2$.  The
Eisenstein bridge gives the extra shift that we need.

There is one important indexing point.  The obstruction to lowering to a
candidate weight $w$ lies in $Q_{w+P}$, not in $Q_w$.  This is proved in
Proposition~\ref{prop:geometric-bockstein}.  The pairing above is therefore a
structural pairing.  It is not yet the pairing of the two actual Bockstein
target spaces.  Theorem~\ref{thm:natural-bockstein-pairing} constructs that
pairing with the correct shift.  This distinction matters: a constant sum of
candidate weights does not by itself prove Bockstein compatibility.
